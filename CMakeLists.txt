cmake_minimum_required(VERSION 3.16)
project(radeon_legacy_vgpu C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(RLG_ENABLE_VULKAN "Build optional Vulkan host backend" ON)
option(RLG_BUILD_TESTS "Build unit tests" ON)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-Wall -Wextra -Wmissing-prototypes -Wstrict-prototypes -Wshadow)
endif()

set(RLG_SOURCES src/chip.c src/device.c src/memory.c src/irq.c src/display.c src/engine2d.c src/cp.c src/mmio.c src/r3d_state.c src/r3d_vertex.c src/r3d_fs.c src/r3d_tex.c src/r3d_raster.c src/r3d_draw.c src/rlg_spirv.c src/vk_shader.c src/vk_backend.c)
add_library(radeonlegacy STATIC ${RLG_SOURCES})
target_include_directories(radeonlegacy PUBLIC include)
if(NOT WIN32)
 target_link_libraries(radeonlegacy PUBLIC m)
endif()
# The Vulkan backend loads the Vulkan loader at run time: only headers are needed.
set(RLG_VULKAN_INCLUDE "" CACHE PATH "Directory containing vulkan/vulkan.h (optional)")
if(RLG_ENABLE_VULKAN)
 find_path(RLG_VK_HEADER_DIR vulkan/vulkan.h HINTS ${RLG_VULKAN_INCLUDE} $ENV{VULKAN_SDK}/include)
 if(RLG_VK_HEADER_DIR)
  target_compile_definitions(radeonlegacy PRIVATE RLG_HAVE_VULKAN=1)
  target_include_directories(radeonlegacy PRIVATE ${RLG_VK_HEADER_DIR})
  target_link_libraries(radeonlegacy PUBLIC ${CMAKE_DL_LIBS})
  message(STATUS "Radeon legacy Vulkan backend enabled (headers: ${RLG_VK_HEADER_DIR})")
 else()
  message(STATUS "Vulkan headers not found; software backend only")
 endif()
endif()
add_executable(radeon-legacy-demo src/demo.c)
target_link_libraries(radeon-legacy-demo PRIVATE radeonlegacy)
if(RLG_BUILD_TESTS)
 enable_testing()
 add_executable(test-core tests/test_core.c)
 target_link_libraries(test-core PRIVATE radeonlegacy)
 add_test(NAME core COMMAND test-core)
 add_executable(test-3d tests/test_3d.c)
 target_link_libraries(test-3d PRIVATE radeonlegacy)
 add_test(NAME 3d COMMAND test-3d)
 add_test(NAME 3d-vulkan COMMAND test-3d vulkan)
 set_tests_properties(3d-vulkan PROPERTIES SKIP_RETURN_CODE 77)
endif()
